(async () => {
    const translations = {
        hu_HU: {
            title: 'Klán Erősítés Összesítő',
            close: 'Bezárás',
            account: 'Fiók',
            total: 'Összesen',
            ownAccount: 'Saját fiók',
            credit: 'Készítette: Patika30',
            wrongPage: 'Ezt a scriptet csak a faluinformáció oldalon lehet futtatni.',
            unitNames: [
                'Lándzsás', 'Kardforgató', 'Bárdos', 'Íjász',
                'Kém', 'Könnyűlovas', 'Lovas íjász', 'Nehézlovas',
                'Faltörő kos', 'Katapult', 'Lovag', 'Főnemes'
            ]
        },
        en_DK: {
            title: 'Tribe Reinforcement Summary',
            close: 'Close',
            account: 'Account',
            total: 'Total',
            ownAccount: 'Own account',
            credit: 'Created by: Patika30',
            wrongPage: 'This script can only be run on the village information screen.',
            unitNames: [
                'Spear fighter', 'Swordsman', 'Axeman', 'Archer',
                'Scout', 'Light cavalry', 'Mounted archer', 'Heavy cavalry',
                'Ram', 'Catapult', 'Paladin', 'Nobleman'
            ]
        },
        en_US: {
            title: 'Tribe Reinforcement Summary',
            close: 'Close',
            account: 'Account',
            total: 'Total',
            ownAccount: 'Own account',
            credit: 'Created by: Patika30',
            wrongPage: 'This script can only be run on the village information screen.',
            unitNames: [
                'Spear fighter', 'Swordsman', 'Axeman', 'Archer',
                'Scout', 'Light cavalry', 'Mounted archer', 'Heavy cavalry',
                'Ram', 'Catapult', 'Paladin', 'Nobleman'
            ]
        }
    };

    const locale = (typeof game_data === 'object' && game_data.locale)
        ? game_data.locale
        : 'hu_HU';

    const t = translations[locale] || translations.en_DK || translations.hu_HU;

    const params = new URLSearchParams(location.search);
    const screen = params.get('screen');
    const rows = document.querySelectorAll('tr[class^="village_row_"]');
    const hasVillageRows = rows.length > 0;

    if (screen !== 'info_village' || !hasVillageRows) {
        alert(t.wrongPage);
        return;
    }

    const unitNames = t.unitNames;

    const ownName = (typeof game_data === 'object' && game_data.player && game_data.player.name)
        ? game_data.player.name.trim()
        : t.ownAccount;

    let playerTotals = {};

    rows.forEach(row => {
        const cells = row.querySelectorAll('td');
        if (cells.length < 2) return;

        const links = cells[0].querySelectorAll('a');
        let playerText = (links.length < 2 || !links[1].textContent.trim())
            ? ownName
            : links[1].textContent.trim();

        const parenStart = playerText.indexOf('(');
        const parenEnd = playerText.indexOf(')');
        let player = '';

        if (parenStart !== -1 && parenEnd !== -1 && parenEnd > parenStart) {
            player = playerText.substring(parenStart + 1, parenEnd).trim();
        } else {
            player = playerText.trim();
        }

        if (!player) return;

        const unitsRaw = Array.from(cells)
            .slice(1, unitNames.length + 1)
            .map(td => parseInt(td.textContent.trim(), 10) || 0);

        if (unitsRaw.every(n => n === 0)) return;

        if (!playerTotals[player]) {
            playerTotals[player] = new Array(unitNames.length).fill(0);
        }

        unitsRaw.forEach((num, idx) => {
            playerTotals[player][idx] += num;
        });
    });

    let existingModal = document.getElementById('clanReinforcementsModal');
    if (existingModal) existingModal.remove();

    let existingOverlay = document.getElementById('clanReinforcementsOverlay');
    if (existingOverlay) existingOverlay.remove();

    const modal = document.createElement('div');
    modal.id = 'clanReinforcementsModal';
    modal.style.position = 'fixed';
    modal.style.top = '50%';
    modal.style.left = '50%';
    modal.style.transform = 'translate(-50%, -50%)';
    modal.style.backgroundColor = '#edd3a1';
    modal.style.border = '2px solid #444';
    modal.style.padding = '20px';
    modal.style.zIndex = 10000;
    modal.style.maxHeight = '80vh';
    modal.style.overflowY = 'auto';
    modal.style.width = '90vw';
    modal.style.boxShadow = '0 0 10px rgba(0,0,0,0.5)';

    let html = `<button id="closeReinforcementsModal" style="float:right; margin-bottom:10px;">${t.close}</button>`;
    html += `<h2 style="margin-top:0">${t.title}</h2>`;
    html += `<table style="border-collapse: collapse; width: 100%;">`;
    html += `<thead><tr><th style="border: 2px solid #fff; padding:5px; text-align:center;">${t.account}</th>`;

    unitNames.forEach(name => {
        html += `<th style="border: 2px solid #fff; padding:5px; text-align:center;">${name}</th>`;
    });

    html += `</tr></thead><tbody>`;

    for (const player in playerTotals) {
        html += `<tr><td style="border: 2px solid #fff; padding:5px; font-weight:bold; text-align:center;">${player}</td>`;
        playerTotals[player].forEach(num => {
            html += `<td style="border: 2px solid #fff; padding:5px; text-align:center;">${num > 0 ? num : ''}</td>`;
        });
        html += `</tr>`;
    }

    const totalUnits = new Array(unitNames.length).fill(0);
    for (const player in playerTotals) {
        playerTotals[player].forEach((num, idx) => {
            totalUnits[idx] += num;
        });
    }

    html += `<tr style="font-weight:bold; background-color:#f0e6c8;">`;
    html += `<td style="border: 2px solid #fff; padding:5px; text-align:center;">${t.total}</td>`;
    totalUnits.forEach(num => {
        html += `<td style="border: 2px solid #fff; padding:5px; text-align:center;">${num > 0 ? num : ''}</td>`;
    });
    html += `</tr>`;

    html += `</tbody></table>`;

    html += `<div style="
        width: 100%;
        display: flex;
        justify-content: flex-end;
        align-items: flex-end;
        margin-top: 8px;
    ">
        <span style="
            font-size: 13px;
            color: #000000;
            opacity: 0.85;
            font-style: italic;
            text-align: right;
        ">
            ${t.credit}
        </span>
    </div>`;

    modal.innerHTML = html;

    modal.querySelector('#closeReinforcementsModal').onclick = () => {
        modal.remove();
        const overlay = document.getElementById('clanReinforcementsOverlay');
        if (overlay) overlay.remove();
    };

    const overlay = document.createElement('div');
    overlay.id = 'clanReinforcementsOverlay';
    overlay.style.position = 'fixed';
    overlay.style.top = 0;
    overlay.style.left = 0;
    overlay.style.width = '100vw';
    overlay.style.height = '100vh';
    overlay.style.backgroundColor = 'rgba(0,0,0,0.3)';
    overlay.style.zIndex = 9999;
    overlay.onclick = () => {
        modal.remove();
        overlay.remove();
    };

    document.body.appendChild(overlay);
    document.body.appendChild(modal);
})();
