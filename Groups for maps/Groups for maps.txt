    async function readPage(response) {
        if (!response.ok) {
            throw new Error(`HTTP ${response.status}`);
        }

        if (new URL(response.url).origin !== location.origin) {
            throw new Error(t("redirect"));
        }

        const page = new DOMParser().parseFromString(
            await response.text(),
            "text/html"
        );

        if (needsVerification(page)) {
            throw new Error(t("login"));
        }

        return page;
    }

    function readVillageFiltersJSON(page, property, required = false) {
        const escapedProperty = property.replace(
            /[.*+?^${}()|[\]\\]/g,
            "\\$&"
        );

        const pattern = new RegExp(
            "\\bVillageFilters\\s*\\.\\s*" +
            escapedProperty +
            "\\s*=\\s*",
            "g"
        );

        for (const script of page.querySelectorAll("script:not([src])")) {
            const source = script.textContent || "";
            pattern.lastIndex = 0;

            let match;

            while ((match = pattern.exec(source))) {
                let start = pattern.lastIndex;

                while (
                    start < source.length &&
                    /\s/.test(source[start])
                ) {
                    start++;
                }

                if (source[start] !== "{" && source[start] !== "[") {
                    continue;
                }

                const stack = [];
                let inString = false;
                let escaped = false;

                for (let index = start; index < source.length; index++) {
                    const character = source[index];

                    if (inString) {
                        if (escaped) {
                            escaped = false;
                        } else if (character === "\\") {
                            escaped = true;
                        } else if (character === '"') {
                            inString = false;
                        }

                        continue;
                    }

                    if (character === '"') {
                        inString = true;
                        continue;
                    }

                    if (character === "{" || character === "[") {
                        stack.push(character);
                        continue;
                    }

                    if (character === "}" || character === "]") {
                        const expected = character === "}" ? "{" : "[";

                        if (stack.pop() !== expected) {
                            throw new Error(
                                t("jsonMalformed", { property })
                            );
                        }

                        if (!stack.length) {
                            try {
                                return JSON.parse(
                                    source.slice(start, index + 1)
                                );
                            } catch {
                                throw new Error(
                                    t("jsonInvalid", { property })
                                );
                            }
                        }
                    }
                }
            }
        }

        if (required) {
            throw new Error(t("jsonMissing", { property }));
        }

        return null;
    }

    function readableFilterValue(value) {
        if (value === null) return "null";

        if (Array.isArray(value)) {
            return "[" +
                value.map(readableFilterValue).join(", ") +
                "]";
        }

        if (typeof value === "object") {
            return Object.entries(value)
                .map(([key, item]) =>
                    `${key.replace(/_/g, " ")}: ${readableFilterValue(item)}`
                )
                .join("; ");
        }

        return String(value);
    }

    function formatAmount(value) {
        const raw = String(value);
        const number = Number(raw);

        return raw.trim() !== "" && Number.isFinite(number)
            ? number.toLocaleString(
                language === "hu" ? "hu-HU" : "en-US"
            )
            : raw;
    }

    function hasFields(value, fields) {
        return (
            value !== null &&
            typeof value === "object" &&
            !Array.isArray(value) &&
            fields.every(field =>
                Object.prototype.hasOwnProperty.call(value, field)
            )
        );
    }

    function formatOperator(value) {
        return operators[value] || String(value);
    }

    function readManualGroupFilters(root = document) {
        const form = root.matches?.("#filter_config")
            ? root
            : root.querySelector("#filter_config");

        if (!form) return [];

        return Array.from(
            form.querySelectorAll(
                'input[type="checkbox"][data-filter="group_static"]'
            )
        )
            .filter(checkbox =>
                checkbox.checked && !checkbox.disabled
            )
            .map(checkbox => {
                const index = checkbox.dataset.id;
                const row = checkbox.closest("tr");

                if (!row) return null;

                const statusSelect = row.querySelector(
                    `select[name="group_static_status_${index}"]`
                );

                const groupSelect = row.querySelector(
                    `select[name="group_static_${index}"]`
                );

                if (
                    !statusSelect ||
                    !groupSelect ||
                    statusSelect.disabled ||
                    groupSelect.disabled
                ) {
                    return null;
                }

                const option =
                    groupSelect.options[groupSelect.selectedIndex];

                if (!option) return null;

                const groupId = groupSelect.value;
                const groupName = option.textContent.trim();
                const status = statusSelect.value;

                let description;

                if (status !== "0" && status !== "1") {
                    return null;
                }

                if (groupId === "0") {
                    description = status === "1"
                        ? t("noManualMembership")
                        : t("hasManualMembership");
                } else {
                    description =
                        `${t(status === "1" ? "memberOf" : "notMemberOf")}: ` +
                        groupName;
                }

                return {
                    filterIndex: index,
                    status,
                    groupId,
                    groupName,
                    description
                };
            })
            .filter(Boolean);
    }

    function formatRule(type, entry, buildings) {
        if (
            type === "building" &&
            hasFields(entry, ["building", "operator", "level"]) &&
            typeof entry.building === "string"
        ) {
            const building =
                buildingNames[entry.building] ||
                buildings[entry.building]?.name ||
                entry.building;

            return (
                `${building} ${formatOperator(entry.operator)} ` +
                `${t("level")} ${entry.level}`
            );
        }

        if (type === "army_type" && hasFields(entry, ["value"])) {
            return armyTypes[entry.value] || String(entry.value);
        }

        if (
            type === "pop_total" &&
            hasFields(entry, ["pop_status", "operator", "pop"])
        ) {
            const populationType =
                populationTypes[entry.pop_status] ||
                String(entry.pop_status);

            return (
                `${populationType} ${formatOperator(entry.operator)} ` +
                formatAmount(entry.pop)
            );
        }

        if (
            type === "incoming_attack" &&
            hasFields(entry, ["has_incoming", "operator", "attacks"])
        ) {
            const hasIncoming = String(entry.has_incoming);

            if (hasIncoming === "0") return t("none");

            if (hasIncoming === "1") {
                return (
                    `${t("attackCount")} ` +
                    `${formatOperator(entry.operator)} ` +
                    formatAmount(entry.attacks)
                );
            }
        }

        if (
            type === "unit" &&
            hasFields(entry, ["unit", "unit_status", "operator", "count"])
        ) {
            const unitName =
                unitNames[entry.unit] || String(entry.unit);

            const unitStatus =
                unitStatuses[entry.unit_status] ||
                String(entry.unit_status);

            return (
                `${unitName} ${unitStatus} ` +
                `${formatOperator(entry.operator)} ` +
                formatAmount(entry.count)
            );
        }

        return readableFilterValue(entry);
    }

    function parseDefinition(page, group, activeFilters) {
        const form = page.querySelector("#filter_config");

        if (!form) {
            throw new Error(t("noDefinition"));
        }

        const returnedId = form.querySelector(
            'input[name="group_id"]'
        )?.value;

        if (returnedId !== String(group.id)) {
            throw new Error(t("wrongGroup"));
        }

        if (
            !activeFilters ||
            typeof activeFilters !== "object" ||
            Array.isArray(activeFilters)
        ) {
            throw new Error(t("badFilterData"));
        }

        const buildings =
            readVillageFiltersJSON(page, "buildings") || {};

        const rules = [];

        for (const [type, entries] of Object.entries(activeFilters)) {
            if (type === "group_static") continue;

            if (!Array.isArray(entries)) {
                throw new Error(t("badFilterType", { type }));
            }

            for (const entry of entries) {
                rules.push({
                    label: labels[type] || type.replace(/_/g, " "),
                    value: formatRule(type, entry, buildings)
                });
            }
        }

        for (const rule of readManualGroupFilters(form)) {
            rules.push({
                label: labels.group_static,
                value: rule.description
            });
        }

        const custom = form.querySelector(
            '[name="use_custom_logic"]'
        )?.value === "1";

        return { rules, custom };
    }

    async function loadInitializedDefinition(group) {
        const frame = document.createElement("iframe");

        frame.title = t("frameTitle");
        frame.setAttribute("aria-hidden", "true");
        frame.tabIndex = -1;

        Object.assign(frame.style, {
            position: "fixed",
            left: "-10000px",
            top: "0",
            width: "1000px",
            height: "700px",
            border: "0",
            pointerEvents: "none",
            opacity: "0"
        });

        let timer = null;

        try {
            return await new Promise((resolve, reject) => {
                let finished = false;
                let loaded = false;
                let lastProblem = t("frameInitializing");

                const deadline = Date.now() + FRAME_TIMEOUT;

                function finish(error, definition) {
                    if (finished) return;

                    finished = true;
                    clearInterval(timer);

                    if (error) reject(error);
                    else resolve(definition);
                }

                frame.addEventListener("load", () => {
                    loaded = true;
                });

                frame.addEventListener("error", () => {
                    finish(new Error(t("frameFailed")));
                });

                timer = setInterval(() => {
                    if (finished) return;

                    if (Date.now() >= deadline) {
                        finish(new Error(t("frameTimeout", {
                            message: lastProblem
                        })));
                        return;
                    }

                    if (!loaded) return;

                    try {
                        const page = frame.contentDocument;

                        if (!page) {
                            lastProblem = t("frameUnavailable");
                            return;
                        }

                        if (page.URL === "about:blank") return;

                        if (needsVerification(page)) {
                            finish(new Error(t("login")));
                            return;
                        }

                        const form = page.querySelector("#filter_config");

                        if (!form) {
                            lastProblem = t("frameNoForm");
                            return;
                        }

                        const returnedId = form.querySelector(
                            '[name="group_id"]'
                        )?.value;

                        if (returnedId !== String(group.id)) {
                            finish(new Error(t("wrongGroup")));
                            return;
                        }

                        const activeFilters = readVillageFiltersJSON(
                            page,
                            "initial_active_filters",
                            true
                        );

                        if (
                            !activeFilters ||
                            typeof activeFilters !== "object" ||
                            Array.isArray(activeFilters)
                        ) {
                            finish(new Error(t("badFilterData")));
                            return;
                        }

                        const expectedManualCount =
                            Array.isArray(activeFilters.group_static)
                                ? activeFilters.group_static.length
                                : 0;

                        if (
                            readManualGroupFilters(form).length <
                            expectedManualCount
                        ) {
                            lastProblem = t("frameManualWaiting");
                            return;
                        }

                        const hasConditions = Object.values(
                            activeFilters
                        ).some(entries =>
                            Array.isArray(entries) && entries.length > 0
                        );

                        const hasControls = Boolean(
                            form.querySelector(
                                'input[type="checkbox"][data-filter]'
                            )
                        );

                        if (hasConditions && !hasControls) {
                            lastProblem = t("frameControlsWaiting");
                            return;
                        }

                        finish(
                            null,
                            parseDefinition(page, group, activeFilters)
                        );
                    } catch (error) {
                        finish(new Error(t("frameReadError", {
                            message: error.message
                        })));
                    }
                }, 250);

                frame.src = filterUrl(group).href;
                document.body.appendChild(frame);
            });
        } finally {
            clearInterval(timer);
            frame.remove();
        }
    }

    function showDefinition(container, definition, group) {
        container.replaceChildren();

        const heading = document.createElement("strong");
        heading.textContent = t("dynamicGroup");

        const logic = document.createElement("div");
        logic.className = "p-logic";
        logic.textContent = t(
            definition.custom ? "customLogic" : "defaultLogic"
        );

        container.append(heading, logic);

        if (definition.rules.length) {
            const ul = document.createElement("ul");

            for (const rule of definition.rules) {
                const li = document.createElement("li");
                li.textContent = `${rule.label}: ${rule.value}`;
                ul.appendChild(li);
            }

            container.appendChild(ul);
        } else {
            const empty = document.createElement("div");
            empty.textContent = t("noConditions");
            container.appendChild(empty);
        }

        appendDefinitionLink(container, group);
    }

    async function toggleDetails(group, details, button) {
        if (busy) return;

        details.hidden = !details.hidden;
        button.textContent =
            `${details.hidden ? "▶" : "▼"} ${group.name}`;

        button.setAttribute(
            "aria-expanded",
            String(!details.hidden)
        );

        if (details.hidden) return;

        if (cache.has(group.id)) {
            showDefinition(details, cache.get(group.id), group);
            return;
        }

        setBusy(true);
        details.textContent = t("loadingDefinition");
        setStatus(t("loadingGroup", { name: group.name }));

        try {
            const definition = await loadInitializedDefinition(group);

            cache.set(group.id, definition);
            showDefinition(details, definition, group);

            setStatus(t("filtersLoaded", { name: group.name }));
        } catch (error) {
            details.replaceChildren();

            const message = document.createElement("div");
            message.textContent = error.message;
            details.appendChild(message);

            appendDefinitionLink(details, group);
            setStatus(t("definitionFailed"), true);

            console.error("[Patika30 filters]", error);
        } finally {
            setBusy(false);
        }
    }

    function updateListState() {
        const query = search.value.trim().toLocaleLowerCase(
            language === "hu" ? "hu-HU" : "en-US"
        );

        let visible = 0;

        list.querySelectorAll(".p-row").forEach(row => {
            const matches = row.dataset.search.includes(query);
            row.hidden = !matches;

            if (matches) visible++;
        });

        get("count").textContent = t(
            query ? "searchCount" : "count",
            {
                visible,
                total: groups.length,
                submitted
            }
        );

        const showMessage = groups.length === 0 || visible === 0;

        listMessage.hidden = !showMessage;

        if (showMessage) {
            listMessage.textContent = t(
                groups.length === 0 ? "empty" : "noResults"
            );
        }
    }

    async function addGroup(group, row) {
        if (busy || !mapForm) return;

        setBusy(true);
        setStatus(t("submitting", { name: group.name }));

        try {
            const action = new URL(
                mapForm.getAttribute("action"),
                mapUrl.href
            );

            if (
                action.origin !== location.origin ||
                action.searchParams.get("action") !== "add_own_group"
            ) {
                throw new Error(t("badMapAction"));
            }

            const body = new URLSearchParams();

            mapForm.querySelectorAll(
                'input[type="hidden"][name]'
            ).forEach(input => {
                if (!input.disabled) {
                    body.append(input.name, input.value);
                }
            });

            body.set("add_group", group.id);

            const response = await fetch(action.href, {
                method: "POST",
                credentials: "same-origin",
                body
            });

            const page = await readPage(response);
            const updatedForm = findMapForm(page);

            if (updatedForm) {
                mapForm = updatedForm.cloneNode(true);
            }

            groups = groups.filter(item => item.id !== group.id);
            submitted++;

            row.remove();
            cache.delete(group.id);
            updateListState();

            setStatus(t("submitted", { name: group.name }));
        } catch (error) {
            setStatus(t("retryWarning", {
                message: error.message
            }), true);

            console.error("[Patika30 add group]", error);
        } finally {
            setBusy(false);
        }
    }

            row.className = "p-row";

            row.dataset.search = group.name.toLocaleLowerCase(
                language === "hu" ? "hu-HU" : "en-US"

            const rowHeader = document.createElement("div");
            rowHeader.className = "p-row-head";

            const expand = document.createElement("button");
            expand.className = "p-expand";
            expand.type = "button";
            expand.textContent = `▶ ${group.name}`;
            expand.setAttribute("aria-expanded", "false");

            const add = document.createElement("button");
            add.className = "p-button p-add";
            add.type = "button";
            add.textContent = t("add");

            const details = document.createElement("div");
            details.className = "p-details";
            details.hidden = true;

            expand.addEventListener("click", () => {
                void toggleDetails(group, details, expand);
            });

            add.addEventListener("click", () => {
                void addGroup(group, row);
            });

            rowHeader.append(expand, add);
            row.append(rowHeader, details);
            list.appendChild(row);
        }

        list.appendChild(listMessage);
        updateListState();
    }

        panel.style.left = `${rect.left}px`;
        panel.style.top = `${rect.top}px`;

        header.setPointerCapture(event.pointerId);
        event.preventDefault();
    });

        if (!drag || drag.id !== event.pointerId) return;

        const maxX = Math.max(0, innerWidth - panel.offsetWidth);
        const maxY = Math.max(0, innerHeight - panel.offsetHeight);

        panel.style.left =
            Math.max(0, Math.min(maxX, event.clientX - drag.x)) + "px";

        panel.style.top =
            Math.max(0, Math.min(maxY, event.clientY - drag.y)) + "px";
    });

    for (const eventName of [
        "pointerup",
        "pointercancel",
        "lostpointercapture"
    ]) {
        header.addEventListener(eventName, () => {
            drag = null;
        });
    }

                credentials: "same-origin",
                cache: "no-store"
            });

            const page = await readPage(response);
            form = findMapForm(page);
        }

        if (!form) {
            throw new Error(t("noMapForm"));
        }

        mapForm = form.cloneNode(true);

        const select = mapForm.querySelector(
            'select[name="add_group"]'
        );

        if (!select) {
            throw new Error(t("noMapSelect"));
        }

        const seen = new Set();

        groups = Array.from(select.options)
            .filter(option => {
                if (
                    !option.value ||
                    option.disabled ||
                    seen.has(option.value)
                ) {
                    return false;
                }

                seen.add(option.value);
                return true;
            })
            .map(option => ({
                id: option.value,
                name: option.textContent.trim()
            }));

        render();
        setStatus(t("ready"));
    } catch (error) {
        setStatus(error.message, true);
        console.error("[Patika30 groups]", error);
    } finally {
        setBusy(false);
    }
})();
