    async function readPage(response) {
        if (!response.ok) {
            throw new Error(`HTTP ${response.status}`);
        }

        if (new URL(response.url).origin !== location.origin) {
            throw new Error("Unexpected redirect.");
        }

        const page = new DOMParser().parseFromString(
            await response.text(),
            "text/html"
        );

        if (page.querySelector(
            'input[type="password"],#bot_check,#botprotection_quest'
        )) {
            throw new Error(
                "Login or verification required. Check the game page."
            );
        }

        return page;
    }

    function readVillageFiltersJSON(page, property, required = false) {
        const escapedProperty = property.replace(
            /[.*+?^${}()|[\]\\]/g,
            "\\$&"
        );

        const pattern = new RegExp(
            "\\bVillageFilters\\s*\\.\\s*" +
            escapedProperty +
            "\\s*=\\s*",
            "g"
        );

        for (const script of page.querySelectorAll(
            "script:not([src])"
        )) {
            const source = script.textContent || "";
            pattern.lastIndex = 0;

            let match;

            while ((match = pattern.exec(source))) {
                let start = pattern.lastIndex;

                while (
                    start < source.length &&
                    /\s/.test(source[start])
                ) {
                    start++;
                }

                if (source[start] !== "{" && source[start] !== "[") {
                    continue;
                }

                const stack = [];
                let inString = false;
                let escaped = false;

                for (
                    let index = start;
                    index < source.length;
                    index++
                ) {
                    const character = source[index];

                    if (inString) {
                        if (escaped) {
                            escaped = false;
                        } else if (character === "\\") {
                            escaped = true;
                        } else if (character === '"') {
                            inString = false;
                        }

                        continue;
                    }

                    if (character === '"') {
                        inString = true;
                        continue;
                    }

                    if (character === "{" || character === "[") {
                        stack.push(character);
                        continue;
                    }

                    if (character === "}" || character === "]") {
                        const expected =
                            character === "}" ? "{" : "[";

                        if (stack.pop() !== expected) {
                            throw new Error(
                                `Malformed JSON data for ${property}.`
                            );
                        }

                        if (!stack.length) {
                            try {
                                return JSON.parse(
                                    source.slice(start, index + 1)
                                );
                            } catch {
                                throw new Error(
                                    `${property} is not valid JSON.`
                                );
                            }
                        }
                    }
                }
            }
        }

        if (required) {
            throw new Error(
                `Saved filter data "${property}" was not found.`
            );
        }

        return null;
    }

    function readableFilterValue(value) {
        if (value === null) return "null";

        if (Array.isArray(value)) {
            return (
                "[" +
                value.map(readableFilterValue).join(", ") +
                "]"
            );
        }

        if (typeof value === "object") {
            return Object.entries(value)
                .map(([key, item]) =>
                    `${key.replace(/_/g, " ")}: ` +
                    readableFilterValue(item)
                )
                .join("; ");
        }

        return String(value);
    }

    function formatAmount(value) {
        const raw = String(value);
        const number = Number(raw);

        return raw.trim() !== "" && Number.isFinite(number)
            ? number.toLocaleString("en-US")
            : raw;
    }

    function hasFields(value, fields) {
        return (
            value !== null &&
            typeof value === "object" &&
            !Array.isArray(value) &&
            fields.every(field =>
                Object.prototype.hasOwnProperty.call(value, field)
            )
        );
    }

    function formatOperator(value) {
        return operators[value] || String(value);

    function readManualGroupFilters(root = document) {
        const form = root.matches?.("#filter_config")
            ? root
            : root.querySelector("#filter_config");

        if (!form) return [];

        return Array.from(
            form.querySelectorAll(
                'input[type="checkbox"][data-filter="group_static"]'
            )
        )
            .filter(checkbox =>
                checkbox.checked && !checkbox.disabled
            )
            .map(checkbox => {
                const index = checkbox.dataset.id;
                const row = checkbox.closest("tr");

                if (!row) return null;

                const statusSelect = row.querySelector(
                    `select[name="group_static_status_${index}"]`
                );

                const groupSelect = row.querySelector(
                    `select[name="group_static_${index}"]`
                );

                if (
                    !statusSelect ||
                    !groupSelect ||
                    statusSelect.disabled ||
                    groupSelect.disabled
                ) {
                    return null;
                }

                const groupOption =
                    groupSelect.options[groupSelect.selectedIndex];

                const statusOption =
                    statusSelect.options[statusSelect.selectedIndex];

                if (!groupOption || !statusOption) return null;

                const relationship =
                    statusOption.textContent.trim() ||
                    (statusSelect.value === "1"
                        ? "Member of"
                        : "Not a member of");

                const groupName = groupOption.textContent.trim();

                return {
                    filterIndex: index,
                    status: statusSelect.value,
                    groupId: groupSelect.value,
                    groupName,
                    description: `${relationship}: ${groupName}`
                };
            })
            .filter(Boolean);
    }

    function formatRule(type, entry, buildings) {
        if (
            type === "building" &&

        ) {
            const building =
                buildings[entry.building]?.name ||
                buildingNames[entry.building] ||
                entry.building;

            return (
                `${building} ${formatOperator(entry.operator)} ` +
                `level ${entry.level}`
            );
        }

        if (
            type === "army_type" &&
            hasFields(entry, ["value"])
        ) {
            return armyTypes[entry.value] || String(entry.value);
        }

        if (
            type === "pop_total" &&
            hasFields(entry, ["pop_status", "operator", "pop"])
        ) {
            const populationType =
                populationTypes[entry.pop_status] ||
                String(entry.pop_status);

            return (
                `${populationType} ${formatOperator(entry.operator)} ` +
                formatAmount(entry.pop)
            );
        }

        if (
            type === "incoming_attack" &&
            hasFields(entry, ["has_incoming", "operator", "attacks"])
        ) {
            const hasIncoming = String(entry.has_incoming);

            if (hasIncoming === "0") return "None";

            if (hasIncoming === "1") {
                return (
                    `Count ${formatOperator(entry.operator)} ` +
                    formatAmount(entry.attacks)
                );
            }
        }

        if (
            type === "unit" &&
            hasFields(entry, [
                "unit",
                "unit_status",
                "operator",
                "count"
            ])
        ) {
            const unitName =
                unitNames[entry.unit] || String(entry.unit);

            const unitStatus =
                unitStatuses[entry.unit_status] ||
                String(entry.unit_status);

            return (
                `${unitName} ${unitStatus} ` +
                `${formatOperator(entry.operator)} ` +
                formatAmount(entry.count)
            );
        }

        return readableFilterValue(entry);
    }

    function parseDefinition(page, group, activeFilters) {
        const form = page.querySelector("#filter_config");

        if (!form) {
            throw new Error(
                "No dynamic filter definition was returned."
            );
        }

        const returnedId = form.querySelector(
            'input[name="group_id"]'
        )?.value;

        if (returnedId !== String(group.id)) {
            throw new Error(
                "The server returned a different group's definition."
            );
        }

        if (
            !activeFilters ||
            typeof activeFilters !== "object" ||
            Array.isArray(activeFilters)
        ) {
            throw new Error("Unexpected saved-filter data format.");
        }

        const buildings =
            readVillageFiltersJSON(page, "buildings") || {};

        const rules = [];

        for (const [type, entries] of Object.entries(activeFilters)) {
            if (type === "group_static") continue;

            if (!Array.isArray(entries)) {
                throw new Error(
                    `Unsupported saved-filter format for "${type}".`
                );
            }

            for (const entry of entries) {
                rules.push({
                    label: labels[type] || type.replace(/_/g, " "),
                    value: formatRule(type, entry, buildings)
                });
            }
        }

        const manualRules = readManualGroupFilters(form);

        for (const rule of manualRules) {
            rules.push({
                label: labels.group_static,
                value: rule.description
            });
        }

        const custom = form.querySelector(
            '[name="use_custom_logic"]'
        )?.value === "1";

        return { rules, custom };
    }

    async function loadInitializedDefinition(group) {
        const frame = document.createElement("iframe");

        frame.title = "Temporary group filter reader";
        frame.setAttribute("aria-hidden", "true");
        frame.tabIndex = -1;

        Object.assign(frame.style, {
            position: "fixed",
            left: "-10000px",
            top: "0",
            width: "1000px",
            height: "700px",
            border: "0",
            pointerEvents: "none",
            opacity: "0"
        });

        let timer = null;

        try {
            return await new Promise((resolve, reject) => {
                let finished = false;
                let loaded = false;
                let lastProblem =
                    "The filter page did not finish initializing.";

                const deadline = Date.now() + FRAME_TIMEOUT;

                function finish(error, definition) {
                    if (finished) return;

                    finished = true;
                    clearInterval(timer);

                    if (error) {
                        reject(error);
                    } else {
                        resolve(definition);
                    }
                }

                frame.addEventListener("load", () => {
                    loaded = true;
                });

                frame.addEventListener("error", () => {
                    finish(new Error(
                        "The filter page could not be loaded in a frame. " +
                        "Open the original filter definition instead."
                    ));
                });

                timer = setInterval(() => {
                    if (finished) return;

                    if (Date.now() >= deadline) {
                        finish(new Error(
                            lastProblem +
                            " The game may block embedded pages. " +
                            "Open the original filter definition to check."
                        ));
                        return;
                    }

                    if (!loaded) return;

                    try {
                        const page = frame.contentDocument;

                        if (!page) {
                            lastProblem =
                                "The embedded document is unavailable.";
                            return;
                        }

                        if (page.URL === "about:blank") return;

                        if (page.querySelector(
                            'input[type="password"],' +
                            '#bot_check,#botprotection_quest'
                        )) {
                            finish(new Error(
                                "Login or verification required. " +
                                "Open the original game page."
                            ));
                            return;
                        }

                        const form =
                            page.querySelector("#filter_config");

                        if (!form) {
                            lastProblem =
                                "No dynamic filter form was found. " +
                                "This may be a manual group or an " +
                                "unsupported page.";
                            return;
                        }

                        const returnedId = form.querySelector(
                            '[name="group_id"]'
                        )?.value;

                        if (returnedId !== String(group.id)) {
                            finish(new Error(
                                "The embedded page returned a " +
                                "different group."
                            ));
                            return;
                        }

                        const activeFilters = readVillageFiltersJSON(
                            page,
                            "initial_active_filters",
                            true
                        );

                        if (
                            !activeFilters ||
                            typeof activeFilters !== "object" ||
                            Array.isArray(activeFilters)
                        ) {
                            finish(new Error(
                                "Unexpected saved-filter data format."
                            ));
                            return;
                        }

                        const expectedManualCount =
                            Array.isArray(activeFilters.group_static)
                                ? activeFilters.group_static.length
                                : 0;

                        const manualRules =
                            readManualGroupFilters(form);

                        if (
                            manualRules.length < expectedManualCount
                        ) {
                            lastProblem =
                                "The embedded page has not rendered " +
                                "all active manual-group dropdowns.";
                            return;
                        }

                        const hasSavedConditions = Object.values(
                            activeFilters
                        ).some(entries =>
                            Array.isArray(entries) && entries.length > 0
                        );

                        const hasRenderedControls = Boolean(
                            form.querySelector(
                                'input[type="checkbox"][data-filter]'
                            )
                        );

                        if (
                            hasSavedConditions &&
                            !hasRenderedControls
                        ) {
                            lastProblem =
                                "The filter controls have not rendered.";
                            return;
                        }

                        const definition = parseDefinition(
                            page,
                            group,
                            activeFilters
                        );

                        finish(null, definition);
                    } catch (error) {
                        finish(new Error(
                            "Could not read the initialized filter page: " +
                            error.message
                        ));
                    }
                }, 250);

                frame.src = filterUrl(group).href;
                document.body.appendChild(frame);
            });
        } finally {
            clearInterval(timer);
            frame.remove();
        }
    }

    function appendDefinitionLink(container, group) {
        const link = document.createElement("a");
        link.href = filterUrl(group).href;
        link.target = "_blank";
        link.rel = "noopener";
        link.textContent = "Open original filter definition »";

        container.appendChild(link);
    }

    function showDefinition(container, definition, group) {
        container.replaceChildren();

        const heading = document.createElement("strong");
        heading.textContent = "Dynamic group";

        const logic = document.createElement("div");
        logic.className = "p-logic";

        logic.textContent = definition.custom
            ? "Custom combinations enabled. Conditions below do not " +
              "show their complete Boolean expression."
            : "All active conditions must match, with the game's OR " +
              "exceptions for listed continents, bonuses and flags.";

        container.append(heading, logic);

        if (definition.rules.length) {
            const ul = document.createElement("ul");

            for (const rule of definition.rules) {
                const li = document.createElement("li");
                li.textContent = `${rule.label}: ${rule.value}`;
                ul.appendChild(li);
            }

            container.appendChild(ul);
        } else {
            const empty = document.createElement("div");
            empty.textContent = "No active conditions were found.";
            container.appendChild(empty);
        }

        appendDefinitionLink(container, group);
    }

    async function toggleDetails(group, details, button) {
        if (busy) return;

        details.hidden = !details.hidden;

        button.textContent =
            `${details.hidden ? "▶" : "▼"} ${group.name}`;

        button.setAttribute(
            "aria-expanded",
            String(!details.hidden)
        );

        if (details.hidden) return;

        if (cache.has(group.id)) {
            showDefinition(details, cache.get(group.id), group);
            return;
        }

        setBusy(true);

        details.textContent =
            "Loading initialized filter page...";

        setStatus(
            `Loading "${group.name}" in a temporary iframe...\n` +
            "The reader does not submit filter changes."
        );

        try {
            const definition =
                await loadInitializedDefinition(group);

            cache.set(group.id, definition);
            showDefinition(details, definition, group);

            setStatus(
                `"${group.name}" filters loaded.\n` +
                "Manual-group descriptions were read from the " +
                "initialized dropdowns. The temporary iframe was removed."
            );
        } catch (error) {
            details.replaceChildren();

            const message = document.createElement("div");
            message.textContent = error.message;

            details.appendChild(message);
            appendDefinitionLink(details, group);

            setStatus(
                "Could not load the definition. " +
                "The temporary iframe was removed. " +
                "No automatic retry was made.",
                true
            );

            console.error("[Patika30 filters]", error);
        } finally {
            setBusy(false);
        }
    }

    function updateCount() {
        get("count").textContent =
            `${groups.length} remaining · ${submitted} submitted`;
    }

    function showEmptyList() {
        if (groups.length) return;

        const message = document.createElement("div");
        message.className = "p-empty";
        message.textContent =
            "No groups remaining in this popup.";

        list.replaceChildren(message);
    }

    function applySearch() {
        const query = search.value.trim().toLocaleLowerCase();

        list.querySelectorAll(".p-row").forEach(row => {
            row.hidden = !row.dataset.search.includes(query);
        });
    }

    async function addGroup(group, row) {
        if (busy || !mapForm) return;

        setBusy(true);
        setStatus(`Submitting "${group.name}"...`);

        try {
            const action = new URL(
                mapForm.getAttribute("action"),
                mapUrl.href
            );

            if (
                action.origin !== location.origin ||
                action.searchParams.get("action") !== "add_own_group"
            ) {
                throw new Error("Unexpected map form action.");
            }

            const body = new URLSearchParams();

            mapForm.querySelectorAll(
                'input[type="hidden"][name]'
            ).forEach(input => {
                if (!input.disabled) {
                    body.append(input.name, input.value);
                }
            });

            body.set("add_group", group.id);

            const response = await fetch(action.href, {
                method: "POST",
                credentials: "same-origin",
                body
            });

            const page = await readPage(response);
            const updatedForm = findMapForm(page);

            if (updatedForm) {
                mapForm = updatedForm.cloneNode(true);
            }

            groups = groups.filter(item =>
                item.id !== group.id
            );

            submitted++;

            row.remove();
            cache.delete(group.id);

            updateCount();
            showEmptyList();

            setStatus(
                `"${group.name}" submitted and removed from this popup.\n` +
                "Persistence has not been independently verified. " +
                "Click Refresh map to check the result."
            );
        } catch (error) {
            setStatus(
                `${error.message}\nNo automatic retry was made. ` +
                "Refresh the map to check before retrying.",
                true
            );

            console.error("[Patika30 add group]", error);
        } finally {
            setBusy(false);
        }
    }

    function render() {
        list.replaceChildren();

        for (const group of groups) {
            const row = document.createElement("div");
            row.className = "p-row";
            row.dataset.search =
                group.name.toLocaleLowerCase();

            const rowHeader = document.createElement("div");
            rowHeader.className = "p-row-head";

            const expand = document.createElement("button");
            expand.className = "p-expand";
            expand.type = "button";
            expand.textContent = `▶ ${group.name}`;
            expand.setAttribute("aria-expanded", "false");

            const add = document.createElement("button");
            add.className = "p-button p-add";
            add.type = "button";
            add.textContent = "Add";

            const details = document.createElement("div");
            details.className = "p-details";
            details.hidden = true;

            expand.addEventListener("click", () => {
                void toggleDetails(group, details, expand);
            });

            add.addEventListener("click", () => {
                void addGroup(group, row);
            });

            rowHeader.append(expand, add);
            row.append(rowHeader, details);
            list.appendChild(row);
        }

        applySearch();
        updateCount();
        showEmptyList();
    }

    search.addEventListener("input", applySearch);

    get("refresh").addEventListener("click", () => {
        if (busy) return;

        setStatus("Reloading the map...");
        get("refresh").disabled = true;

        location.reload();
    });

    get("close").addEventListener("click", () => {
        if (busy) {
            setStatus(
                "Wait for the current request to finish. " +
                "Filter loading times out after 20 seconds."
            );
            return;
        }

        panel.remove();
        style.remove();
        cache.clear();

        delete window[KEY];
    });

    const header = get("header");
    let drag = null;

    header.addEventListener("pointerdown", event => {
        if (
            event.button !== 0 ||
            event.target.closest("button")
        ) {
            return;
        }

        const rect = panel.getBoundingClientRect();

        drag = {
            id: event.pointerId,
            x: event.clientX - rect.left,
            y: event.clientY - rect.top
        };

        panel.style.transform = "none";
        panel.style.left = `${rect.left}px`;
        panel.style.top = `${rect.top}px`;

        header.setPointerCapture(event.pointerId);
        event.preventDefault();
    });

    header.addEventListener("pointermove", event => {
        if (!drag || drag.id !== event.pointerId) return;

        const maxX = Math.max(
            0,
            innerWidth - panel.offsetWidth
        );

        const maxY = Math.max(
            0,
            innerHeight - panel.offsetHeight
        );

        panel.style.left =
            Math.max(
                0,
                Math.min(maxX, event.clientX - drag.x)
            ) + "px";

        panel.style.top =
            Math.max(
                0,
                Math.min(maxY, event.clientY - drag.y)
            ) + "px";
    });

    for (const eventName of [
        "pointerup",
        "pointercancel",
        "lostpointercapture"
    ]) {
        header.addEventListener(eventName, () => {
            drag = null;
        });
    }

    setBusy(true);
    setStatus("Loading map group list...");

    try {
        let form = findMapForm(document);

        if (!form) {
            const response = await fetch(mapUrl.href, {
                credentials: "same-origin",
                cache: "no-store"
            });

            const page = await readPage(response);
            form = findMapForm(page);
        }

        if (!form) {
            throw new Error(
                'Open "Add own group" on the map, ' +
                "then close this popup and run the script again."
            );
        }

        mapForm = form.cloneNode(true);

        const select = mapForm.querySelector(
            'select[name="add_group"]'
        );

        if (!select) {
            throw new Error("Map group dropdown not found.");
        }

        const seen = new Set();

        groups = Array.from(select.options)
            .filter(option => {
                if (
                    !option.value ||
                    option.disabled ||
                    seen.has(option.value)
                ) {
                    return false;
                }

                seen.add(option.value);
                return true;
            })
            .map(option => ({
                id: option.value,
                name: option.textContent.trim()
            }));

        render();

        setStatus(
            "Click a group name to read its initialized filters.\n" +
            "Only Add submits a map-highlight change. " +
            "Refresh map reloads the game page."
        );
    } catch (error) {
        setStatus(error.message, true);
        console.error("[Patika30 groups]", error);
    } finally {
        setBusy(false);
    }
})();
